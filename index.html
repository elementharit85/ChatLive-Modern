Setup Tabel SQL Supabase

Jalankan perintah SQL ini sekali saja di menu SQL Editor pada dashboard Supabase Anda (`zyddldgtefrtwjgymbtk`).

-- 1. Tabel User dengan Nomor 4-Digit & PIN
CREATE TABLE IF NOT EXISTS public.chat_users (
    phone_4 VARCHAR(4) PRIMARY KEY,
    pin VARCHAR(10) NOT NULL,
    display_name TEXT NOT NULL,
    avatar_url TEXT,
    is_online BOOLEAN DEFAULT true,
    last_seen TIMESTAMPTZ DEFAULT NOW(),
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. Tabel Kontak (Invite Nomor 4-Digit)
CREATE TABLE IF NOT EXISTS public.chat_contacts (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    owner_phone VARCHAR(4) NOT NULL,
    contact_phone VARCHAR(4) NOT NULL,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(owner_phone, contact_phone)
);

-- 3. Tabel Pesan Live Chat (Teks & Gambar Base64, Tanpa Video)
CREATE TABLE IF NOT EXISTS public.chat_messages (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    sender_phone VARCHAR(4) NOT NULL,
    receiver_phone VARCHAR(4) NOT NULL,
    message_text TEXT,
    image_data TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 4. Matikan RLS (Atau Buka Akses Public Anon) Agar Langsung Jalan
ALTER TABLE public.chat_users DISABLE ROW LEVEL SECURITY;
ALTER TABLE public.chat_contacts DISABLE ROW LEVEL SECURITY;
ALTER TABLE public.chat_messages DISABLE ROW LEVEL SECURITY;

-- 5. Aktifkan Supabase Realtime untuk Ketiga Tabel
ALTER PUBLICATION supabase_realtime ADD TABLE public.chat_users;
ALTER PUBLICATION supabase_realtime ADD TABLE public.chat_contacts;
ALTER PUBLICATION supabase_realtime ADD TABLE public.chat_messages;

Setelah klik Run di SQL Editor Supabase, klik tombol Cek Ulang Koneksi.
